# 声学异常检测模块实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 ESP32-P4 金属件检测项目实现可插拔的声学异常检测模块：设备端提取 Mel 谱图并用小型 CNN Autoencoder 推理，PC 端提供训练与模型导出脚本。

**Architecture:** 设备端新增 `app_audio_features`（Mel 谱图）和 `app_audio_ai`（Autoencoder 推理+校准）两个模块；PC 端用 PyTorch 训练 Autoencoder 并导出 espdl；模块通过 `audio_ai_infer()` 暴露异常分数，供后续视觉模块融合。

**Tech Stack:** ESP-IDF 5.x, esp-dl 3.x, esp-dsp, C/C++, Python 3, PyTorch, ONNX

## Task 2: 生成 Mel 滤波器组头文件

**Files:**
- Create: `tools/generate_mel_filterbank.py`
- Create: `main/app_audio_mel_fb.h`

- [ ] **Step 1: 编写生成脚本**

```python
import numpy as np
import argparse
import os

SAMPLE_RATE = 16000
FFT_SIZE = 512
NUM_MEL_BINS = 32
FMIN = 0
FMAX = 8000
NUM_FREQ_BINS = FFT_SIZE // 2 + 1  # 257

def hz_to_mel(hz):
    return 2595.0 * np.log10(1.0 + hz / 700.0)

def mel_to_hz(mel):
    return 700.0 * (10.0 ** (mel / 2595.0) - 1.0)

def build_mel_filterbank():
    mel_min = hz_to_mel(FMIN)
    mel_max = hz_to_mel(FMAX)
    mel_points = np.linspace(mel_min, mel_max, NUM_MEL_BINS + 2)
    hz_points = mel_to_hz(mel_points)

    bin_indices = np.floor((FFT_SIZE + 1) * hz_points / SAMPLE_RATE).astype(int)
    filterbank = np.zeros((NUM_MEL_BINS, NUM_FREQ_BINS), dtype=np.float32)

    for i in range(NUM_MEL_BINS):
        for j in range(bin_indices[i], bin_indices[i + 1]):
            filterbank[i, j] = (j - bin_indices[i]) / (bin_indices[i + 1] - bin_indices[i])
        for j in range(bin_indices[i + 1], bin_indices[i + 2]):
            filterbank[i, j] = (bin_indices[i + 2] - j) / (bin_indices[i + 2] - bin_indices[i + 1])

    return filterbank

def emit_c_header(filterbank, out_path):
    rows, cols = filterbank.shape
    lines = [
        "/* Auto-generated by tools/generate_mel_filterbank.py */",
        "#ifndef APP_AUDIO_MEL_FB_H",
        "#define APP_AUDIO_MEL_FB_H",
        "",
        f"#define AUDIO_MEL_FB_ROWS {rows}",
        f"#define AUDIO_MEL_FB_COLS {cols}",
        "",
        "static const float g_mel_filterbank[AUDIO_MEL_FB_ROWS][AUDIO_MEL_FB_COLS] = {"
    ]
    for i in range(rows):
        row_vals = ", ".join(f"{v:.8f}f" for v in filterbank[i])
        lines.append(f"    {{{row_vals}}},")
    lines.extend([
        "};",
        "",
        "#endif // APP_AUDIO_MEL_FB_H"
    ])
    with open(out_path, "w") as f:
        f.write("\n".join(lines))

if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--output", default="main/app_audio_mel_fb.h")
    args = parser.parse_args()

    fb = build_mel_filterbank()
    emit_c_header(fb, args.output)
    print(f"Generated {args.output}: shape={fb.shape}")
```

```c
#include "app_audio_features.h"
#include "app_audio_def.h"
#include "app_audio_mel_fb.h"
#include "esp_dsp.h"
#include "esp_log.h"
#include <math.h>
#include <string.h>

static const char *TAG = "AUDIO_FEATURES";

static float s_hann_window[AUDIO_FFT_SIZE];
static float s_fft_buffer[AUDIO_FFT_SIZE * 2];
static float s_frame[AUDIO_FFT_SIZE];
static float s_power[AUDIO_FFT_SIZE / 2 + 1];

esp_err_t audio_features_init(void)
{
    dsps_wind_hann_f32(s_hann_window, AUDIO_FFT_SIZE);
    // FFT engine is initialized by the existing mic_fft_task using
    // dsps_fft2r_init_fc32(). We only need the Hann window here.
    return ESP_OK;
}

static void compute_power_spectrum(const int16_t *pcm, uint32_t sample_count,
                                   int frame_idx, float *power)
{
    int start = frame_idx * AUDIO_HOP_SIZE;
    for (int i = 0; i < AUDIO_FFT_SIZE; i++) {
        int idx = start + i;
        float v = (idx < (int)sample_count) ? (float)pcm[idx] : 0.0f;
        s_frame[i] = v * s_hann_window[i];
    }

    for (int i = 0; i < AUDIO_FFT_SIZE; i++) {
        s_fft_buffer[i * 2 + 0] = s_frame[i];
        s_fft_buffer[i * 2 + 1] = 0.0f;
    }

    dsps_fft2r_fc32(s_fft_buffer, AUDIO_FFT_SIZE);
    dsps_bit_rev_fc32(s_fft_buffer, AUDIO_FFT_SIZE);

    int num_bins = AUDIO_FFT_SIZE / 2 + 1;
    for (int i = 0; i < num_bins; i++) {
        float re = s_fft_buffer[i * 2 + 0];
        float im = s_fft_buffer[i * 2 + 1];
        power[i] = (re * re + im * im) / (float)AUDIO_FFT_SIZE;
    }
}

static void apply_mel_filterbank(const float *power, float *mel)
{
    for (int m = 0; m < AUDIO_NUM_MEL_BINS; m++) {
        float sum = 0.0f;
        for (int f = 0; f < AUDIO_MEL_FB_COLS; f++) {
            sum += power[f] * g_mel_filterbank[m][f];
        }
        mel[m] = sum;
    }
}

static void log_compress(float *mel)
{
    for (int m = 0; m < AUDIO_NUM_MEL_BINS; m++) {
        mel[m] = logf(mel[m] + 1e-6f);
    }
}

esp_err_t audio_features_extract_mel(const int16_t *pcm, uint32_t sample_count,
                                     float *mel_out, size_t mel_out_len)
{
    if (mel_out_len < AUDIO_NUM_FRAMES * AUDIO_NUM_MEL_BINS) {
        return ESP_ERR_INVALID_SIZE;
    }
    if (sample_count < AUDIO_TOTAL_SAMPLES) {
        return ESP_ERR_INVALID_ARG;
    }

    float mel_min = 1e30f;
    float mel_max = -1e30f;

    // First pass: compute all mel frames and find min/max for normalization
    for (int t = 0; t < AUDIO_NUM_FRAMES; t++) {
        int frame_idx = t * 2;  // downsample by 2
        compute_power_spectrum(pcm, sample_count, frame_idx, s_power);
        apply_mel_filterbank(s_power, &mel_out[t * AUDIO_NUM_MEL_BINS]);
        log_compress(&mel_out[t * AUDIO_NUM_MEL_BINS]);

        for (int m = 0; m < AUDIO_NUM_MEL_BINS; m++) {
            float v = mel_out[t * AUDIO_NUM_MEL_BINS + m];
            if (v < mel_min) mel_min = v;
            if (v > mel_max) mel_max = v;
        }
    }

    // Normalize to [0, 1]
    float scale = mel_max - mel_min;
    if (scale < 1e-6f) scale = 1.0f;

    for (int i = 0; i < AUDIO_NUM_FRAMES * AUDIO_NUM_MEL_BINS; i++) {
        mel_out[i] = (mel_out[i] - mel_min) / scale;
    }

    return ESP_OK;
}
```

## Task 4: 创建 PC 端 Mel 谱图可视化脚本

**Files:**
- Create: `tools/convert_raw_to_mel.py`

- [ ] **Step 1: 编写脚本**

```python
import numpy as np
import matplotlib.pyplot as plt
import argparse
import os

SAMPLE_RATE = 16000
FFT_SIZE = 512
HOP_SIZE = 256
NUM_MEL_BINS = 32
FMIN = 0
FMAX = 8000
NUM_FRAMES = 32


def hz_to_mel(hz):
    return 2595.0 * np.log10(1.0 + hz / 700.0)


def mel_to_hz(mel):
    return 700.0 * (10.0 ** (mel / 2595.0) - 1.0)


def build_mel_filterbank():
    mel_min = hz_to_mel(FMIN)
    mel_max = hz_to_mel(FMAX)
    mel_points = np.linspace(mel_min, mel_max, NUM_MEL_BINS + 2)
    hz_points = mel_to_hz(mel_points)

    bin_freqs = np.linspace(0, SAMPLE_RATE / 2, FFT_SIZE // 2 + 1)
    fb = np.zeros((NUM_MEL_BINS, FFT_SIZE // 2 + 1), dtype=np.float32)

    for i in range(NUM_MEL_BINS):
        left_hz = hz_points[i]
        center_hz = hz_points[i + 1]
        right_hz = hz_points[i + 2]
        for j in range(FFT_SIZE // 2 + 1):
            f = bin_freqs[j]
            if f <= left_hz or f >= right_hz:
                continue
            elif f < center_hz:
                fb[i, j] = (f - left_hz) / (center_hz - left_hz)
            else:
                fb[i, j] = (right_hz - f) / (right_hz - center_hz)
    return fb


def load_raw(path):
    data = np.fromfile(path, dtype=np.int16)
    if len(data) < 16000:
        data = np.pad(data, (0, 16000 - len(data)))
    return data[:16000]


def extract_mel(pcm):
    window = np.hanning(FFT_SIZE)
    fb = build_mel_filterbank()

    frames = []
    for t in range(NUM_FRAMES):
        start = t * 2 * HOP_SIZE
        frame = pcm[start:start + FFT_SIZE].astype(np.float32) * window
        fft = np.fft.rfft(frame, n=FFT_SIZE)
        power = (np.abs(fft) ** 2) / FFT_SIZE
        mel = fb @ power
        mel = np.log(mel + 1e-6)
        frames.append(mel)

    mel_spec = np.stack(frames, axis=0)  # [32, 32]
    mel_spec = (mel_spec - mel_spec.min()) / (mel_spec.max() - mel_spec.min() + 1e-6)
    return mel_spec


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--input", required=True, help="Path to .raw audio file")
    parser.add_argument("--output", default=None, help="Optional output image path")
    parser.add_argument("--npy", default=None, help="Optional output .npy path")
    args = parser.parse_args()

    pcm = load_raw(args.input)
    mel = extract_mel(pcm)

    print(f"Mel shape: {mel.shape}, min={mel.min():.4f}, max={mel.max():.4f}")

    if args.npy:
        np.save(args.npy, mel[..., np.newaxis].astype(np.float32))
        print(f"Saved {args.npy}")

    plt.figure(figsize=(8, 4))
    plt.imshow(mel.T, aspect="auto", origin="lower", cmap="viridis")
    plt.colorbar(label="Normalized log-Mel")
    plt.xlabel("Frame")
    plt.ylabel("Mel bin")
    plt.title(f"Mel spectrogram: {os.path.basename(args.input)}")
    plt.tight_layout()

    if args.output:
        plt.savefig(args.output)
        print(f"Saved plot to {args.output}")
    else:
        plt.show()


if __name__ == "__main__":
    main()
```

```python
import numpy as np
import argparse
import os
from pathlib import Path

from convert_raw_to_mel import load_raw, extract_mel


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--input", required=True, help="Directory containing good/*.raw")
    parser.add_argument("--output", required=True, help="Output .npy file")
    parser.add_argument("--max-samples", type=int, default=None)
    args = parser.parse_args()

    input_dir = Path(args.input)
    files = sorted(input_dir.glob("*.raw"))
    if args.max_samples:
        files = files[:args.max_samples]

        raise ValueError(f"No .raw files found in {input_dir}")

    for f in files:
        pcm = load_raw(str(f))
        mel = extract_mel(pcm)
        mels.append(mel[..., np.newaxis])
        print(f"Processed {f.name}")

    mels = np.stack(mels, axis=0).astype(np.float32)
    print(f"Dataset shape: {mels.shape}")
    np.save(args.output, mels)
    print(f"Saved {args.output}")


if __name__ == "__main__":
    main()
```

## Task 6: 创建 Autoencoder 训练脚本

**Files:**
- Create: `tools/train_audio_autoencoder.py`

- [ ] **Step 1: 编写训练脚本**

```python
import torch
import torch.nn as nn
import numpy as np
import argparse
import json
import os

NUM_FRAMES = 32
NUM_MEL_BINS = 32
LATENT_DIM = 16


class AudioAutoencoder(nn.Module):
    def __init__(self):
        super().__init__()
        self.encoder = nn.Sequential(
            nn.Conv2d(1, 8, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(2),
            nn.Conv2d(8, 4, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(2),
            nn.Flatten(),
        )
        self.fc_encode = nn.Linear(4 * 8 * 8, LATENT_DIM)

        self.fc_decode = nn.Linear(LATENT_DIM, 4 * 8 * 8)
        self.decoder = nn.Sequential(
            nn.Unflatten(1, (4, 8, 8)),
            nn.Upsample(scale_factor=2, mode="nearest"),
            nn.Conv2d(4, 8, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.Upsample(scale_factor=2, mode="nearest"),
            nn.Conv2d(8, 4, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.Conv2d(4, 1, kernel_size=3, padding=1),
            nn.Sigmoid(),
        )

    def forward(self, x):
        z = self.encoder(x)
        z = self.fc_encode(z)
        z = self.fc_decode(z)
        out = self.decoder(z)
        return out


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--data", required=True, help="Path to good_mels.npy")
    parser.add_argument("--output", default="models/audio_autoencoder", help="Output directory")
    parser.add_argument("--epochs", type=int, default=200)
    parser.add_argument("--batch-size", type=int, default=32)
    parser.add_argument("--lr", type=float, default=1e-3)
    args = parser.parse_args()

    os.makedirs(args.output, exist_ok=True)

    data = np.load(args.data).astype(np.float32)  # [N, 32, 32, 1]
    data = np.transpose(data, (0, 3, 1, 2))        # [N, 1, 32, 32] for PyTorch (NCHW)
    dataset = torch.from_numpy(data)
    loader = torch.utils.data.DataLoader(dataset, batch_size=args.batch_size, shuffle=True)

    model = AudioAutoencoder()
    criterion = nn.MSELoss()
    optimizer = torch.optim.Adam(model.parameters(), lr=args.lr)

    model.train()
    for epoch in range(args.epochs):
        total_loss = 0.0
        for batch in loader:
            optimizer.zero_grad()
            recon = model(batch)
            loss = criterion(recon, batch)
            loss.backward()
            optimizer.step()
            total_loss += loss.item() * batch.size(0)
        avg_loss = total_loss / len(dataset)
        if (epoch + 1) % 20 == 0:
            print(f"Epoch {epoch + 1}/{args.epochs}, loss={avg_loss:.6f}")

    # Compute threshold statistics on training data
    model.eval()
    with torch.no_grad():
        recon = model(dataset)
        mse = ((recon - dataset) ** 2).mean(dim=(1, 2, 3)).numpy()

    mean = float(np.mean(mse))
    std = float(np.std(mse))
    k = 3.0

    print(f"Threshold stats: mean={mean:.6f}, std={std:.6f}, k={k}")

    torch.save(model.state_dict(), os.path.join(args.output, "audio_autoencoder.pth"))

    with open(os.path.join(args.output, "threshold.json"), "w") as f:
        json.dump({"mean": mean, "std": std, "k": k}, f, indent=2)

    # Save ONNX in NCHW form; esp-dl converter handles transpose
    dummy = torch.randn(1, 1, NUM_FRAMES, NUM_MEL_BINS)
    torch.onnx.export(
        model,
        dummy,
        os.path.join(args.output, "audio_autoencoder.onnx"),
        input_names=["input"],
        output_names=["output"],
        opset_version=11,
        dynamic_axes={"input": {0: "batch"}, "output": {0: "batch"}}
    )
    print(f"Saved ONNX and checkpoint to {args.output}")


if __name__ == "__main__":
    main()
```

```python
import argparse
import os
import sys


def find_espdl_converter():
    idf_path = os.environ.get("IDF_PATH", "")
    candidates = [
        os.path.join(idf_path, "components", "esp-dl", "tools", "onnx_to_espdl.py"),
        os.path.join(idf_path, "components", "esp-dl", "tools", "conversion", "onnx_to_espdl.py"),
    ]
    # Also check managed_components
    proj_root = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
    candidates.append(os.path.join(proj_root, "managed_components", "espressif__esp-dl", "tools", "onnx_to_espdl.py"))
    for c in candidates:
        if os.path.exists(c):
            return c
    return None


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--onnx", required=True, help="Path to audio_autoencoder.onnx")
    parser.add_argument("--output", required=True, help="Output .espdl path")
    args = parser.parse_args()

    converter = find_espdl_converter()
    if converter is None:
        print("ERROR: Could not find esp-dl onnx_to_espdl.py converter.")
        print("Please set IDF_PATH or ensure esp-dl is in managed_components.")
        sys.exit(1)

    cmd = f"python {converter} --input {args.onnx} --output {args.output}"
    print(f"Running: {cmd}")
    ret = os.system(cmd)
    if ret != 0:
        print("espdl conversion failed")
        sys.exit(1)
    print(f"Generated {args.output}")


if __name__ == "__main__":
    main()
```

```cpp
#include "app_audio_ai.h"
#include "app_audio_def.h"
#include "app_audio_features.h"
#include "dl_tensor_base.hpp"
#include "dl_model_base.hpp"
#include "esp_log.h"
#include "cJSON.h"
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

static const char *TAG = "AUDIO_AI";

static dl::Model *s_model = nullptr;
static std::map<std::string, dl::TensorBase*> s_input_map;
static std::string s_input_name;
static float s_mel_buffer[AUDIO_AI_INPUT_H * AUDIO_AI_INPUT_W * AUDIO_AI_INPUT_C];
static float s_threshold_mean = 0.0f;
static float s_threshold_std = 1.0f;
static float s_threshold_k = AUDIO_AI_DEFAULT_K;

static float s_good_mse[AUDIO_AI_MAX_GOOD_SAMPLES];
static int s_good_mse_count = 0;

esp_err_t audio_ai_init(const void *model_addr)
{
    if (s_model != nullptr) {
        return ESP_OK;
    }

    ESP_LOGI(TAG, "Loading audio autoencoder model...");
    s_model = new dl::Model((const char *)model_addr, fbs::MODEL_LOCATION_IN_FLASH_RODATA);
    if (!s_model) {
        ESP_LOGE(TAG, "Failed to load audio model");
        return ESP_FAIL;
    }

    auto &inputs = s_model->get_inputs();
    if (inputs.empty()) {
        ESP_LOGE(TAG, "Model has no inputs");
        return ESP_FAIL;
    }

    s_input_name = inputs.begin()->first;
    s_input_map[s_input_name] = inputs.begin()->second;

    auto shape = inputs.begin()->second->get_shape();
    ESP_LOGI(TAG, "Audio model input shape: [%d,%d,%d,%d]",
             shape[0], shape[1], shape[2], shape[3]);

    return audio_features_init();
}

static esp_err_t run_inference(float *mel_input, float *recon_output)
{
    if (!s_model) {
        return ESP_ERR_INVALID_STATE;
    }

    dl::TensorBase *input_tensor = s_input_map[s_input_name];
    if (input_tensor->get_dtype() != dl::DATA_TYPE_FLOAT) {
        ESP_LOGE(TAG, "Model input must be float");
        return ESP_ERR_NOT_SUPPORTED;
    }

    memcpy(input_tensor->get_element_ptr(), mel_input,
           AUDIO_AI_INPUT_H * AUDIO_AI_INPUT_W * AUDIO_AI_INPUT_C * sizeof(float));

    s_model->run(s_input_map);

    auto &outputs = s_model->get_outputs();
    if (outputs.empty()) {
        return ESP_FAIL;
    }

    dl::TensorBase *out_tensor = outputs.begin()->second;
    memcpy(recon_output, out_tensor->get_element_ptr(),
           AUDIO_AI_INPUT_H * AUDIO_AI_INPUT_W * AUDIO_AI_INPUT_C * sizeof(float));

    return ESP_OK;
}

static float compute_mse(const float *a, const float *b, int len)
{
    float sum = 0.0f;
    for (int i = 0; i < len; i++) {
        float d = a[i] - b[i];
        sum += d * d;
    }
    return sum / (float)len;
}

esp_err_t audio_ai_infer(const int16_t *pcm, uint32_t sample_count,
                         float *anomaly_score, float *recon_mse)
{
    if (!pcm || !anomaly_score) {
        return ESP_ERR_INVALID_ARG;
    }

    esp_err_t ret = audio_features_extract_mel(pcm, sample_count, s_mel_buffer,
                                               sizeof(s_mel_buffer) / sizeof(float));
    if (ret != ESP_OK) {
        return ret;
    }

    float recon[AUDIO_AI_INPUT_H * AUDIO_AI_INPUT_W * AUDIO_AI_INPUT_C];
    ret = run_inference(s_mel_buffer, recon);
    if (ret != ESP_OK) {
        return ret;
    }

    float mse = compute_mse(s_mel_buffer, recon,
                            AUDIO_AI_INPUT_H * AUDIO_AI_INPUT_W * AUDIO_AI_INPUT_C);
    if (recon_mse) {
        *recon_mse = mse;
    }

    if (s_threshold_std < 1e-6f) {
        *anomaly_score = 0.0f;
    } else {
        *anomaly_score = (mse - s_threshold_mean) / s_threshold_std;
    }

    ESP_LOGI(TAG, "MSE=%.6f, anomaly_score=%.3f", mse, *anomaly_score);
    return ESP_OK;
}

esp_err_t audio_ai_learn_good_sample(const int16_t *pcm, uint32_t sample_count)
{
    if (s_good_mse_count >= AUDIO_AI_MAX_GOOD_SAMPLES) {
        ESP_LOGW(TAG, "Good sample buffer full");
        return ESP_ERR_NO_MEM;
    }

    float score = 0.0f;
    float mse = 0.0f;
    esp_err_t ret = audio_ai_infer(pcm, sample_count, &score, &mse);
    if (ret != ESP_OK) {
        return ret;
    }

    s_good_mse[s_good_mse_count++] = mse;
    ESP_LOGI(TAG, "Learned good sample %d, mse=%.6f", s_good_mse_count, mse);
    return ESP_OK;
}

esp_err_t audio_ai_finalize_threshold(void)
{
    if (s_good_mse_count < 5) {
        ESP_LOGE(TAG, "Need at least 5 good samples, got %d", s_good_mse_count);
        return ESP_ERR_INVALID_STATE;
    }

    float sum = 0.0f;
    for (int i = 0; i < s_good_mse_count; i++) {
        sum += s_good_mse[i];
    }
    float mean = sum / s_good_mse_count;

    float sq_sum = 0.0f;
    for (int i = 0; i < s_good_mse_count; i++) {
        float d = s_good_mse[i] - mean;
        sq_sum += d * d;
    }
    float std = sqrtf(sq_sum / s_good_mse_count);

    s_threshold_mean = mean;
    s_threshold_std = std;
    ESP_LOGI(TAG, "Threshold finalized: mean=%.6f, std=%.6f", mean, std);
    return ESP_OK;
}

esp_err_t audio_ai_save_threshold(const char *path)
{
    if (!path) return ESP_ERR_INVALID_ARG;

    cJSON *root = cJSON_CreateObject();
    cJSON_AddNumberToObject(root, "mean", s_threshold_mean);
    cJSON_AddNumberToObject(root, "std", s_threshold_std);
    cJSON_AddNumberToObject(root, "k", s_threshold_k);

    char *str = cJSON_Print(root);
    FILE *f = fopen(path, "w");
    if (!f) {
        cJSON_free(str);
        cJSON_Delete(root);
        return ESP_FAIL;
    }
    fprintf(f, "%s", str);
    fclose(f);

    cJSON_free(str);
    cJSON_Delete(root);
    ESP_LOGI(TAG, "Threshold saved to %s", path);
    return ESP_OK;
}

esp_err_t audio_ai_load_threshold(const char *path)
{
    if (!path) return ESP_ERR_INVALID_ARG;

    FILE *f = fopen(path, "r");
    if (!f) {
        ESP_LOGW(TAG, "Threshold file not found: %s", path);
        return ESP_FAIL;
    }

    fseek(f, 0, SEEK_END);
    long len = ftell(f);
    fseek(f, 0, SEEK_SET);
    char *buf = (char *)malloc(len + 1);
    if (!buf) {
        fclose(f);
        return ESP_ERR_NO_MEM;
    }
    fread(buf, 1, len, f);
    buf[len] = '\0';
    fclose(f);

    cJSON *root = cJSON_Parse(buf);
    free(buf);
    if (!root) {
        return ESP_FAIL;
    }

    cJSON *mean = cJSON_GetObjectItem(root, "mean");
    cJSON *std = cJSON_GetObjectItem(root, "std");
    cJSON *k = cJSON_GetObjectItem(root, "k");

    if (mean) s_threshold_mean = (float)mean->valuedouble;
    if (std) s_threshold_std = (float)std->valuedouble;
    if (k) s_threshold_k = (float)k->valuedouble;

    cJSON_Delete(root);
    ESP_LOGI(TAG, "Threshold loaded: mean=%.6f, std=%.6f, k=%.2f",
             s_threshold_mean, s_threshold_std, s_threshold_k);
    return ESP_OK;
}

esp_err_t audio_ai_get_threshold(float *mean, float *std, float *k)
{
    if (mean) *mean = s_threshold_mean;
    if (std) *std = s_threshold_std;
    if (k) *k = s_threshold_k;
    return ESP_OK;
}
```
